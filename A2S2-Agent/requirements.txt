langchain

langchain-openai        # provider: any OpenAI-compatible endpoint (vLLM, LM Studio, ...)
requests                # thin client for the A2S2-OS HTTP API
mcp                     # MCP server: expose the OS tools to Claude Desktop/Code & other clients
